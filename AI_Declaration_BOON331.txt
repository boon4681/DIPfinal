const points = [
  { x: 10, y: 20 },
  { x: 40, y: 15 },
  { x: 5,  y: 80 },
  { x: 30, y: 45 }
];

function getBoundingBox(points) {
  if (points.length === 0) return null;

  let minX = points[0].x;
  let maxX = points[0].x;
  let minY = points[0].y;
  let maxY = points[0].y;

  for (let i = 1; i < points.length; i++) {
    const p = points[i];
    if (p.x < minX) minX = p.x;
    if (p.x > maxX) maxX = p.x;
    if (p.y < minY) minY = p.y;
    if (p.y > maxY) maxY = p.y;
  }

  return {
    x: minX,
    y: minY,
    width: maxX - minX,
    height: maxY - minY,
    maxX: maxX,
    maxY: maxY
  };
}

console.log(getBoundingBox(points));
